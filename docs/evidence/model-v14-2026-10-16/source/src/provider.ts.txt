      const { value, done } = await reader.read(); if (done) break;
      bytes += value.byteLength;
      if (bytes > maxBytes) { await reader.cancel(); throw new Error('Provider response exceeds size limit'); }
      chunks.push(value);
    }
    return JSON.parse(Buffer.concat(chunks).toString('utf8'));
  } finally { reader.releaseLock(); }
}
export async function providerAnswer(config: { url: string; key: string; model: string }, prompt: keyof typeof promptTemplates, question: string, fetcher: typeof fetch = fetch) {
  const url = new URL(config.url);
  if (url.protocol !== 'https:' || url.username || url.password) throw new Error('Provider requires HTTPS without URL credentials');
  const snapshot = promptSnapshot(prompt);
  const responseMode = (prompt === 'v8' || prompt === 'v9' || prompt === 'v10' || prompt === 'v11' || prompt === 'v12' || prompt === 'v13' || prompt === 'v14') ? responseModeV8(question) : undefined;
  const policyContext = prompt === 'v12' ? policyContextV12(question) : undefined;
  const business = prompt === 'v13' || prompt === 'v14' ? decisionContextV1(question) : undefined;
  const contextualText = business ? snapshot.text + '\n\n' + business.text : policyContext ? snapshot.text + '\n\n' + policyContext.text : snapshot.text;
  const systemText = responseMode ? contextualText + '\n\n' + (prompt === 'v14' ? responseModesV14 : responseModesV8)[responseMode] : contextualText;
  const response = await fetcher(url, {
    method: 'POST', headers: { authorization: 'Bearer ' + config.key, 'content-type': 'application/json' },
    signal: AbortSignal.timeout(snapshot.requestTimeoutMs), redirect: 'error',
    body: JSON.stringify({ model: config.model, messages: [{ role: 'system', content: systemText }, { role: 'user', content: question }], max_completion_tokens: snapshot.maxOutputTokens, ...('temperature' in snapshot ? { temperature: snapshot.temperature } : {}), ...('reasoningEffort' in snapshot ? { reasoning_effort: snapshot.reasoningEffort } : {}) }),
  });
  if (!response.ok) { await response.body?.cancel(); throw new Error('Provider HTTP ' + response.status); }
  const body = await boundedJson(response);
  const output = body?.choices?.[0]?.message?.content;
  // Capture only bounded, known metadata. Provider error bodies may contain private data.
  const usage: Record<string, number> = {};
  for (const field of ['prompt_tokens', 'completion_tokens', 'total_tokens']) {
    const n = body?.usage?.[field]; if (Number.isSafeInteger(n) && n >= 0) usage[field] = n;
  }
  const finishReason = typeof body?.choices?.[0]?.finish_reason === 'string' ? body.choices[0].finish_reason.slice(0, 80) : null;
  const valid = typeof output === 'string' && output.trim().length > 0 && output.length <= 10000;
  const error = finishReason === 'length' ? 'Provider answer truncated at token limit; incomplete evidence' : !valid ? 'Provider response contract invalid: nonempty text answer required (max 10000 characters)' : finishReason && finishReason !== 'stop' ? 'Provider did not finish a text answer: ' + finishReason : undefined;
  const guard = business ? (prompt === 'v14' ? checkExplanationV2 : checkExplanationV1)(typeof output === 'string' ? output : undefined, business.decision, responseMode ?? 'plain', error ?? (finishReason !== 'stop' ? 'Incomplete provider completion metadata' : undefined)) : undefined;
  return { ...(typeof output === 'string' && output.length <= 10000 ? { output } : {}), ...(error ? { error } : {}), metadata: { ...(business ? { businessDecision: business.decision, guard, decisionContextSha256: createHash('sha256').update(business.text).digest('hex') } : {}), ...(policyContext ? { policyContextVersion: policyContext.version, policyTopics: policyContext.topics, policyContextSha256: createHash('sha256').update(policyContext.text).digest('hex') } : {}), ...(responseMode ? { responseMode, systemPromptSha256: createHash('sha256').update(systemText).digest('hex') } : {}), requestId: typeof body?.id === 'string' ? body.id.slice(0, 200) : null, reportedModel: typeof body?.model === 'string' ? body.model.slice(0, 200) : null, finishReason, usage, httpStatus: response.status, contentType: typeof output, contentCharacters: typeof output === 'string' ? output.length : null } };
}
